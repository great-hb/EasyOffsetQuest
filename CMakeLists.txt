cmake_minimum_required(VERSION 3.22)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
project(EasyOffsetQuest VERSION 0.1.6 LANGUAGES C CXX)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(EasyOffsetQuest SHARED ${SOURCES})
set(COMPILE_ID EasyOffsetQuest)
set(EXTERN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/extern)
function(RECURSE_FILES output pattern)
    file(GLOB_RECURSE result ${pattern})
    list(FILTER result EXCLUDE REGEX "\\.debug\\.so$")
    set(${output} ${result} PARENT_SCOPE)
endfunction()
target_include_directories(EasyOffsetQuest PRIVATE include)
include(${CMAKE_CURRENT_SOURCE_DIR}/extern.cmake)
target_compile_definitions(EasyOffsetQuest PRIVATE MOD_ID="EasyOffsetQuest" VERSION="0.1.6" QUEST)
target_compile_options(EasyOffsetQuest PRIVATE -frtti -fexceptions -fvisibility=hidden -fPIC)
# Keep machine-specific workspace paths out of __FILE__ and debug metadata.
target_compile_options(EasyOffsetQuest PRIVATE
    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=EasyOffsetQuest"
    "-fdebug-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=EasyOffsetQuest")
target_link_libraries(EasyOffsetQuest PRIVATE log)
target_link_options(EasyOffsetQuest PRIVATE -Wl,--as-needed -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
file(GLOB_RECURSE INLINE_HOOK_SOURCES extern/includes/beatsaber-hook/shared/inline-hook/*.cpp extern/includes/beatsaber-hook/shared/inline-hook/*.c)
target_sources(EasyOffsetQuest PRIVATE ${INLINE_HOOK_SOURCES})
